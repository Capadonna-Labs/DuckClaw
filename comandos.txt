-------- PARA LEVANTAR EL SISTEMA --------

# Día cero (primera vez en esta máquina):
install.cmd                                # Windows — doble clic
.\install.cmd                              # Windows PowerShell (obligatorio el .\)
cmd /c install.cmd                         # Windows PowerShell alternativa
./duckops-up.sh                            # macOS / Linux / WSL

# Deploy completo y limpio del framework (UN comando):
#   uv sync + migrate + PM2 recycle + espejo vault + estado Kiwix
cd <raíz-del-repo>
uv run duckops stack deploy --full

# Variantes:
uv run duckops stack deploy                # sync + migrate + PM2 (sin mirror)
uv run duckops stack deploy --no-migrate   # solo reinicio PM2
uv run duckops stack deploy --no-sync      # omitir uv sync (deploy rápido)

# Utilidades:
uv run duckops up                          # wizard / arranque interactivo
uv run duckops doctor --strict
uv run duckops knowledge status            # vault mirror + ZIMs
uv run duckops knowledge mirror            # solo re-sync GDrive → local
uv run duckops configure
uv run duckops down --all

# Document toolbox (ofimática local — MarkItDown + Report Engine + LibreOffice PDF):
# Python: uv sync  (markitdown/docxtpl en duckclaw-shared; NO hay extra document-toolbox)
brew install --cask libreoffice            # export_docx_to_pdf (Word → PDF)
uv run duckops doctor                      # Check MarkItDown / LibreOffice / OUTPUT_ROOTS
# Flujo: Report Engine → .docx en OUTPUT → export_docx_to_pdf → .pdf al lado

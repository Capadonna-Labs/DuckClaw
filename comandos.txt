-------- PARA LEVANTAR EL SISTEMA --------

source .venv/bin/activate                  # activa venv (opcional; uv run no lo exige)
uv sync                                    # deps Python del monorepo (también lo hace duckops up)
uv run duckops doctor --strict             # health check antes de probar
uv run duckops configure                   # si es primera vez o cambiaste .env
uv run duckops up                          # uv sync + gateway + redis + servicios

uv run duckops down --all               
uv run duckops stack down               

--------- INTERFAZ GRAFICA ----------
cd apps/duckclaw-admin
pnpm install                             
pnpm dev
